## Usage Examples

### Example 1: Send 2FA Code Email

```typescript
import { sendTwoFactorCodeEmail } from '@/lib/email/send';

const result = await sendTwoFactorCodeEmail(
  'user@example.com',
  'Max Mustermann',
  '123456', // 6-digit code
  5, // expires in 5 minutes
  'de' // German locale
);

if (result.success) {
  console.log('2FA code email sent:', result.messageId);
} else {
  console.error('Failed to send 2FA email:', result.error);
}
```

### Example 2: Send Account Deletion Scheduled Email

```typescript
import { sendAccountDeletionScheduledEmail } from '@/lib/email/send';

const deletionDate = new Date();
deletionDate.setDate(deletionDate.getDate() + 30);

const formattedDate = deletionDate.toLocaleDateString('de-DE', {
  day: '2-digit',
  month: 'long',
  year: 'numeric'
}); // "15. Dezember 2025"

const cancelUrl = `${process.env.NEXTAUTH_URL}/business/settings/account?action=cancel-deletion`;

const result = await sendAccountDeletionScheduledEmail(
  'user@example.com',
  'Max Mustermann',
  formattedDate,
  cancelUrl,
  'de'
);
```

### Example 3: Send Account Deletion Confirmed Email

```typescript
import { sendAccountDeletionConfirmedEmail } from '@/lib/email/send';

// Send BEFORE deleting the user from database
const result = await sendAccountDeletionConfirmedEmail(
  user.email,
  user.name || 'User',
  'de'
);

if (result.success) {
  // Now safe to delete user
  await prisma.user.delete({ where: { id: user.id } });
}
```
